const SHEET_URL = "https://docs.google.com/spreadsheets/d/1qVNJPR90XAMNPBsnyHkCLgj5xDWOnccno1XJMWmD09Y/edit?gid=645769912#gid=645769912"; const SHEET_NAME = "Students"; function doGet(e) { let result; try { const action = String(e.parameter.action || "list").toLowerCase(); if (action === "list") { result = listStudents(); } else if (action === "set") { result = setStudent(e.parameter); } else { result = { success: false, message: "Unknown action." }; } } catch (error) { result = { success: false, message: error.message }; } return jsonpResponse(result, e.parameter.callback); } function listStudents() { const sheet = getSheet(); ensureSheetStructure(sheet); const lastRow = sheet.getLastRow(); if (lastRow < 2) { return { success: true, students: [] }; } const values = sheet.getRange(2, 1, lastRow - 1, 2).getValues(); const students = values .filter(row => String(row[0] || "").trim() !== "") .map(row => ({ name: String(row[0] || "").trim(), hearts: Number(row[1]) || 0 })); return { success: true, students: students }; } function setStudent(params) { const studentName = String(params.student || "").trim(); const hearts = Math.max(0, Math.floor(Number(params.hearts))); if (!studentName) { return { success: false, message: "Missing student name." }; } if (!Number.isFinite(hearts)) { return { success: false, message: "Invalid heart number." }; } const sheet = getSheet(); ensureSheetStructure(sheet); const lastRow = sheet.getLastRow(); let foundRow = -1; if (lastRow >= 2) { const names = sheet.getRange(2, 1, lastRow - 1, 1).getValues(); const targetName = studentName.toLowerCase(); for (let i = 0; i < names.length; i++) { const currentName = String(names[i][0] || "").trim().toLowerCase(); if (currentName === targetName) { foundRow = i + 2; break; } } } if (foundRow === -1) { sheet.appendRow([studentName, hearts]); } else { sheet.getRange(foundRow, 2).setValue(hearts); } return { success: true, message: "Saved.", student: studentName, hearts: hearts }; } function getSheet() { if (!SHEET_URL || SHEET_URL.includes("PASTE_YOUR")) { throw new Error("Add your Google Sheet URL inside SHEET_URL."); } const spreadsheet = SpreadsheetApp.openByUrl(SHEET_URL); let sheet = spreadsheet.getSheetByName(SHEET_NAME); if (!sheet) { sheet = spreadsheet.insertSheet(SHEET_NAME); } return sheet; } function ensureSheetStructure(sheet) { const lastRow = sheet.getLastRow(); if (lastRow === 0) { sheet.getRange(1, 1, 1, 2).setValues([["Student", "Hearts"]]); return; } const headers = sheet.getRange(1, 1, 1, 2).getValues()[0]; const firstHeader = String(headers[0] || "").trim(); const secondHeader = String(headers[1] || "").trim(); if (!firstHeader && !secondHeader) { sheet.getRange(1, 1, 1, 2).setValues([["Student", "Hearts"]]); return; } if (firstHeader !== "Student" || secondHeader !== "Hearts") { throw new Error('Your sheet must use these exact headers: Student | Hearts'); } } function jsonpResponse(data, callback) { const safeCallback = String(callback || "callback"); if (!/^[a-zA-Z_$][0-9a-zA-Z_$]*$/.test(safeCallback)) { return ContentService .createTextOutput(JSON.stringify({ success: false, message: "Invalid callback name." })) .setMimeType(ContentService.MimeType.JSON); } return ContentService .createTextOutput(`${safeCallback}(${JSON.stringify(data)});`) .setMimeType(ContentService.MimeType.JAVASCRIPT); }
